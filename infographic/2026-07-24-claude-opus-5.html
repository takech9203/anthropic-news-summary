コードサンプル

Python エフォートレベルを指定した使用例 
import anthropic

client = anthropic.Anthropic()

# エフォートレベルを指定した使用例
response = client.messages.create(
    model="claude-opus-5",
    max_tokens=16384,
    thinking={"type": "enabled", "budget_tokens": 10000},
    effort="max",
    messages=[
        {"role": "user", "content": "複雑な推論問題を解いてください。"}
    ]
)
